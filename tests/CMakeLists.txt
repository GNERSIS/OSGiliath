# ── GTest test suite ─────────────────────────────────────────
# Style-matched to the l0 template: include(AddGTest) first, then
# one add_executable / target_link_libraries / add_test per test.
# ─────────────────────────────────────────────────────────────

include(AddGTest)

# The osg target propagates ${PROJECT_SOURCE_DIR}/include via its PUBLIC
# BUILD_INTERFACE. The CMake-generated headers (osg/Config, osg/GL,
# osg/Version) live under ${PROJECT_BINARY_DIR}/include and are added to the
# directory include path by OsgConfigHeaders at the top level, which
# propagates here. We additionally name both explicitly per test target so
# the tests build standalone regardless of directory-property inheritance.
set(OSG_TEST_INCLUDE_DIRS
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_BINARY_DIR}/include
)

# ── maths conventions ──
add_executable(maths_conventions_test maths_conventions_test.cpp)
target_link_libraries(maths_conventions_test PRIVATE osg gtest_main)
target_include_directories(maths_conventions_test PRIVATE ${OSG_TEST_INCLUDE_DIRS})
add_test(NAME maths_conventions_test COMMAND maths_conventions_test)

# ── object model ──
add_executable(object_model_test object_model_test.cpp)
target_link_libraries(object_model_test PRIVATE osg gtest_main)
target_include_directories(object_model_test PRIVATE ${OSG_TEST_INCLUDE_DIRS})
add_test(NAME object_model_test COMMAND object_model_test)

# ── all headers compile standalone-in-sequence ──
# Generate one #include line per public osg header into a single TU, then
# compile it under the full strict toolchain (-Werror). The value of this
# test is that the translation unit compiles at all.
#
# Excluded (with reasons):
#   maths/[A-Z]*.hpp            — dead/legacy/redirect capitalized headers;
#                                 e.g. Matrixd.hpp redefines lowercase live
#                                 types (class vec3 / dmat4) → ODR clash.
#   rendering/HeadlessCapture.hpp — drags in EGL + osgViewer.
#   core/InheritAcceptImpl.hpp    — include-order-sensitive impl detail
#                                   (must follow NodeVisitor definition).
file(GLOB_RECURSE OSG_ALL_HEADERS
    RELATIVE ${CMAKE_SOURCE_DIR}/include
    CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/include/osg/*.hpp
)
list(FILTER OSG_ALL_HEADERS EXCLUDE REGEX "^osg/maths/[A-Z].*\\.hpp$")
list(FILTER OSG_ALL_HEADERS EXCLUDE REGEX "^osg/rendering/HeadlessCapture\\.hpp$")
list(FILTER OSG_ALL_HEADERS EXCLUDE REGEX "^osg/core/InheritAcceptImpl\\.hpp$")
list(SORT OSG_ALL_HEADERS)

set(OSG_ALL_HEADERS_CONTENT
    "/* OSGiliath — OpenSceneGraph fork. See LICENSE.txt */\n"
    "/* Generated by tests/CMakeLists.txt — do not edit. */\n"
    "#pragma once\n"
)
foreach(_hdr IN LISTS OSG_ALL_HEADERS)
    list(APPEND OSG_ALL_HEADERS_CONTENT "#include <${_hdr}>\n")
endforeach()
string(JOIN "" OSG_ALL_HEADERS_FILE_BODY ${OSG_ALL_HEADERS_CONTENT})
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/all_osg_headers.hpp "${OSG_ALL_HEADERS_FILE_BODY}")

add_executable(all_headers_test all_headers_test.cpp)
target_link_libraries(all_headers_test PRIVATE osg gtest_main)
target_include_directories(all_headers_test PRIVATE
    ${OSG_TEST_INCLUDE_DIRS}
    ${CMAKE_CURRENT_BINARY_DIR}
)
add_test(NAME all_headers_test COMMAND all_headers_test)
